startup page for members.bellinghammakerspace.org

sudo useradd -m  maker
sudo passwd -d maker
sudo usermod -s /sbin/nologin maker
sudo nano /etc/gdm3/custom.conf
	AutomaticLoginEnable = true
	AutomaticLogin = maker
	TimedLoginEnable = true 
	TimedLogin = maker 
	TimedLoginDelay = 5
